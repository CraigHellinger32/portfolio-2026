Style Guide - Funnelback search

This Funnelback Search Style Guide will provide you with all the markup, styles and front-end script components of your search results page.

Dependencies

In all of the markup below we are referencing classes and IDs from your own CSS as well as Funnelback's. The following files are needed in addition to the below styles and markup:

Client

  • Bootstrap has been removed. Bolton's implementation will use Spacecraft's proprietary framework Orbit.

Funnelback

  • Font Awesome has been removed. Bolton use their own custom icon set
  • jQuery (https://ajax.googleapis.com/ajax/libs/jquery/1.11.2/jquery.min.js)
  • Bootstrap JS (files/bootstrap.min.js) - This has been removed.
  • Typeahead (files/typeahead.js) - This has been packaged with Orbit.js
  • Handlebars (files/multichannel/handlebars.js) - This has been packaged with Orbit.js
  • Autocomplete Templates (dist/libs/typeahead.autocomplete.templates.fb.js)
  • Autocomplete (dist/libs/typeahead.autocomplete.fb.js)

Datepicker

  • Datepicker has been removed as there is no instance of it being used in the styleguide.

// Bloodhound setup
var organic =  new Bloodhound({
   datumTokenizer: Bloodhound.tokenizers.obj.whitespace('key')
  ,queryTokenizer: Bloodhound.tokenizers.whitespace
  ,limit: 6
  ,local : suggest
});

var events =  new Bloodhound({
     datumTokenizer: Bloodhound.tokenizers.obj.whitespace('key')
    ,queryTokenizer: Bloodhound.tokenizers.whitespace
    ,limit: 3
    ,local: suggestevents
});

var news =  new Bloodhound({
     datumTokenizer: Bloodhound.tokenizers.obj.whitespace('key')
    ,queryTokenizer: Bloodhound.tokenizers.whitespace
    ,limit: 3
    ,local: suggestpeople
});

// Initialise
organic.initialize();
events.initialize();
news.initialize();

$siteHeader.find("#search-site").add("#query-mc").typeahead({
    minLength : 1,
    hint: true,
    highlight: true
  }
 ,{
   name : "organic"
   ,displayKey : "value"
   ,source : organic.ttAdapter()
   ,templates : {
       empty : Handlebars.compile(empty__organic)
      ,header : Handlebars.compile(header__organic)
      ,suggestion : Handlebars.compile(result__organic)
     }
   }
  ,{
    name : "events"
   ,displayKey : "value"
   ,source : events.ttAdapter()
   ,templates : {
       empty : Handlebars.compile(empty__events)
      ,header : Handlebars.compile(header__events)
      ,suggestion : Handlebars.compile(result__events)
     }
   }
  ,{
    name : "news"
   ,displayKey : "value"
   ,source : news.ttAdapter()
   ,templates : {
       empty : Handlebars.compile(empty__news)
      ,header : Handlebars.compile(header__news)
      ,suggestion : Handlebars.compile(result__news)
     }
   }
).on('typeahead:selected',function($e, datum){
        // Debug only
        console.log(datum);
        // For entries with a URL
        if(datum.url){
          location = datum.url;
          return false;
        }
  // We never fire a direct search
});
});

// TYPEAHEAD TEMPLATES
